cmake_minimum_required(VERSION 3.20)
project(BigLeagueBowling VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
foreach(suite rules campaign)
 add_executable(bowling-${suite}-tests tests/${suite}.cpp)
 target_include_directories(bowling-${suite}-tests PRIVATE src)
 add_test(NAME native-${suite} COMMAND bowling-${suite}-tests)
endforeach()
set(RAYLIB_ROOT "" CACHE PATH "Optional extracted official raylib 5.5 SDK")
if(RAYLIB_ROOT)
 add_library(raylib STATIC IMPORTED)
 set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${RAYLIB_ROOT}/lib/libraylib.a" INTERFACE_INCLUDE_DIRECTORIES "${RAYLIB_ROOT}/include")
 if(WIN32)
  target_link_libraries(raylib INTERFACE opengl32 gdi32 winmm)
 elseif(UNIX AND NOT APPLE)
  find_library(OPENGL_RUNTIME NAMES GL libGL.so.1 REQUIRED)
  find_library(X11_RUNTIME NAMES X11 libX11.so.6 REQUIRED)
  target_link_libraries(raylib INTERFACE ${OPENGL_RUNTIME} ${X11_RUNTIME} m pthread dl rt)
 endif()
else()
 include(FetchContent)
 set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
 FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(raylib)
endif()
add_executable(big-league-bowling src/main.cpp)
target_link_libraries(big-league-bowling PRIVATE raylib)
if(MSVC)
 target_compile_options(big-league-bowling PRIVATE /W4)
else()
 target_compile_options(big-league-bowling PRIVATE -Wall -Wextra)
endif()
if(WIN32)
 set_target_properties(big-league-bowling PROPERTIES WIN32_EXECUTABLE TRUE OUTPUT_NAME "Big League Bowling")
endif()
add_custom_command(TARGET big-league-bowling POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:big-league-bowling>/assets")
install(TARGETS big-league-bowling RUNTIME DESTINATION .)
install(DIRECTORY assets DESTINATION .)
